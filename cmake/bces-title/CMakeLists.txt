cmake_minimum_required(VERSION 3.24)
project(Infamous2BCESTitle LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    message(FATAL_ERROR "This archive is for the separately configured iPhoneOS target")
endif()
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SDK "${PORT_ROOT}/upstream/infamous2-recomp/ps3recomp-fork")
set(GENERATED "${PORT_ROOT}/build/bces01229/recompiled")
file(GLOB PPU CONFIGURE_DEPENDS "${GENERATED}/ppu_recomp_*.cpp")
list(LENGTH PPU COUNT)
if(NOT COUNT EQUAL 164 OR NOT EXISTS "${GENERATED}/ppu_recomp.h")
    message(FATAL_ERROR "Prepare the complete verified BCES PPU source set first")
endif()
add_library(bces_title STATIC ${PPU})
target_compile_features(bces_title PRIVATE cxx_std_20)
target_include_directories(bces_title PRIVATE "${GENERATED}" "${SDK}/include" "${SDK}/runtime/ppu")
target_compile_options(bces_title PRIVATE -O1 -w -fno-strict-aliasing -fwrapv -ffp-contract=off)
message(STATUS "iPhoneOS BCES title archive: ${COUNT} real PPU chunks, no macOS libraries or app runner")
